   118→
   119→function YtPlayerHost() {
   120→  const hostRef = useRef<HTMLDivElement>(null);
   121→  const created = useRef(false);
   122→
   123→  useEffect(() => {
   124→    if (created.current) return;
   125→    created.current = true;
   126→
   127→    const tryCreate = () => {
   128→      if (!window.YT || !hostRef.current || !window.YT.Player) return false;
   129→      try {
   130→        const p = new window.YT.Player(hostRef.current, {
   131→          height: "180",
   132→          width: "320",
   133→          playerVars: {
   134→            autoplay: 0,
   135→            controls: 0,
   136→            disablekb: 1,
   137→            modestbranding: 1,
   138→            playsinline: 1,
   139→            rel: 0,
   140→            origin: window.location.origin,
   141→          },
   142→          events: {
   143→            onReady: () => {
   144→              (window as unknown as { __ytPlayer?: unknown }).__ytPlayer = p;
   145→              attachYtPlayer(p as never);
   146→            },
   147→            onStateChange: (e) => handlePlayerStateChange(e.data),
   148→            onError: (e) => handlePlayerError((e as { data?: number })?.data),
   149→          },
   150→        });
   151→        return true;
   152→      } catch {
   153→        return false;
   154→      }
   155→    };
   156→
   157→    if (!tryCreate()) {
   158→      if (!document.getElementById("yt-iframe-api")) {
   159→        const tag = document.createElement("script");
   160→        tag.id = "yt-iframe-api";
   161→        tag.src = "https://www.youtube.com/iframe_api";
   162→        document.head.appendChild(tag);
   163→      }
   164→      const prev = window.onYouTubeIframeAPIReady;
   165→      window.onYouTubeIframeAPIReady = () => {
   166→        prev?.();
   167→        tryCreate();
   168→      };
   169→    }
   170→  }, []);
   171→
   172→  return (
   173→    <div className="fixed -left-[9999px] top-0 w-[320px] h-[180px] overflow-hidden pointer-events-none" aria-hidden>
